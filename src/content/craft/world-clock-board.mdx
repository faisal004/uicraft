## Implementation

Copy this component. Each row has five separate blocks for `HH:MM`, a clear gap, and 15 separate blocks for the city. The 5×7 letter patterns live in the file, so there is no font or extra stylesheet to load. City names longer than 15 characters are clipped to fit.

```tsx title="world-clock-board.tsx"
"use client";

import { useEffect, useRef, useState } from "react";

const FONT: Record<string, string> = {

  "0": "01110100011001110101110011000101110",
  "1": "00100011000010000100001000010001110",
  "2": "01110100010000100010001000100011111",

  F: "11111100001000011110100001000010000",

  J: "00111000100001000010000101001001100",
  K: "10001100101010011000101001001010001",
  L: "10000100001000010000100001000011111",
  M: "10001110111010110101100011000110001",
  N: "10001110011010110011100011000110001",

  Q: "01110100011000110001101011001001101",
  R: "11110100011000111110101001001010001",
  S: "01111100001000001110000010000111110",
  T: "11111001000010000100001000010000100",
  U: "10001100011000110001100011000101110",
  V: "10001100011000110001010100010000100",
  W: "10001100011000110101101011010101010",
  X: "10001100010101000100010101000110001",
  Y: "10001100010101000100001000010000100",
  Z: "11111000010001000100010001000011111",
};

const CHARSET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
const CELL_W = 5;
const CELL_H = 7;
const TIME_SLOTS = 5;
const CITY_SLOTS = 15;
const HOLD_MS = 320;

const DEFAULT_CITIES = [
  { name: "NEW YORK CITY", timeZone: "America/New_York" },
  { name: "TOKYO", timeZone: "Asia/Tokyo" },
  { name: "HAWAII", timeZone: "Pacific/Honolulu" },
  { name: "LOS ANGELES", timeZone: "America/Los_Angeles" },
  { name: "NEW DELHI", timeZone: "Asia/Kolkata" },
];

type City = {
  name: string;
  timeZone: string;
};

type WorldClockBoardProps = {
  cities?: City[];
  staggerMs?: number;
  cycleMs?: number;
  replay?: number;
  announce?: boolean;
  className?: string;
};

function formatTime(timeZone: string, date: Date) {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone,
    hour: "2-digit",
    minute: "2-digit",
    hourCycle: "h23",
  }).formatToParts(date);
  const hour = parts.find((part) => part.type === "hour")?.value ?? "00";
  const minute = parts.find((part) => part.type === "minute")?.value ?? "00";
  return `${hour.slice(-2).padStart(2, "0")}:${minute.slice(-2).padStart(2, "0")}`;
}

function clockLines(cities: City[], date: Date | null) {
  return cities.map((city) => {
    const time = date ? formatTime(city.timeZone, date) : " ".repeat(TIME_SLOTS);
    const name = (city.name.trim().toUpperCase() || "CITY").slice(0, CITY_SLOTS);
    return `${time}${name.padEnd(CITY_SLOTS, " ")}`;
  });
}

function lit(character: string, x: number, y: number) {
  const glyph = FONT[character] ?? FONT[" "];
  return glyph[y * CELL_W + x] === "1";
}

function LedMatrix({ lines }: { lines: string[] }) {
  return (
    <div
      className="grid w-full gap-y-[clamp(14px,2.4cqi,28px)]"
    >
      {lines.map((line, rowIndex) => (
        <div
          key={rowIndex}
          className="grid items-start gap-x-[clamp(4px,0.7cqi,8px)]"
          style={{ gridTemplateColumns: `repeat(${TIME_SLOTS}, minmax(0, 1fr)) minmax(0, 1fr) repeat(${CITY_SLOTS}, minmax(0, 1fr))` }}
        >
          {[...line].map((character, column) => (
            <div
              key={column}
              className="grid grid-cols-5 gap-[clamp(1px,0.16cqi,2px)]"
              style={column === TIME_SLOTS ? { gridColumn: column + 2 } : undefined}
            >
              {Array.from({ length: CELL_W * CELL_H }, (_, dot) => (
                <span
                  key={dot}
                  className={`aspect-square w-full rounded-full ${lit(character, dot % CELL_W, Math.floor(dot / CELL_W)) ? "bg-[#f2f2f2]" : "bg-[#434343]"}`}
                />
              ))}
            </div>
          ))}
        </div>
      ))}
    </div>
  );
}

function useFlapRows(rows: string[], staggerMs: number, cycleMs: number, replay: number) {
  const signature = rows.join("\n");
  const [shown, setShown] = useState(() => rows.map((row) => " ".repeat(row.length)));
  const previous = useRef<string[] | null>(null);
  const replaySeen = useRef(replay);

  useEffect(() => {
    const next = signature.length === 0 ? [] : signature.split("\n");
    const stagger = Math.min(160, Math.max(12, staggerMs));
    const cycle = Math.min(180, Math.max(32, cycleMs));
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const prior = previous.current;
    const force = replay !== replaySeen.current;

    const dirty = next.map((row, rowIndex) =>
      [...row].map((character, column) => {
        if (reduce) return false;
        if (force || prior === null) return true;
        if (column >= TIME_SLOTS && prior[rowIndex]?.slice(0, TIME_SLOTS) !== row.slice(0, TIME_SLOTS)) return true;
        return prior[rowIndex]?.[column] !== character;
      }),
    );

    if (!dirty.some((row) => row.some(Boolean))) {
      previous.current = next;
      replaySeen.current = replay;
      setShown(next);
      return;
    }

    const started = performance.now();
    let timer = 0;

    const paint = () => {
      const now = performance.now();
      let pending = false;
      const frame = next.map((row, rowIndex) => {
        let line = "";
        for (let column = 0; column < row.length; column += 1) {
          const goal = row[column] ?? " ";
          const city = column >= TIME_SLOTS;
          const position = city ? column - TIME_SLOTS : column;
          const last = city ? CITY_SLOTS - 1 : TIME_SLOTS - 1;
          const delay = stagger * last * (1 - Math.cos(Math.PI * position / last)) / 2;
          if (!dirty[rowIndex]?.[column] || now >= started + delay + HOLD_MS) {
            line += goal;
          } else if (now < started + delay) {
            pending = true;
            line += prior?.[rowIndex]?.[column] ?? " ";
          } else {
            pending = true;
            line += CHARSET[Math.floor(Math.random() * CHARSET.length)];
          }
        }
        return line;
      });
      setShown(frame);
      if (!pending) {
        previous.current = next;
        replaySeen.current = replay;
        window.clearInterval(timer);
      }
      return pending;
    };

    if (paint()) timer = window.setInterval(paint, cycle);
    return () => window.clearInterval(timer);
  }, [signature, staggerMs, cycleMs, replay]);

  return shown;
}

export function WorldClockBoard({
  cities = DEFAULT_CITIES,
  staggerMs = 70,
  cycleMs = 80,
  replay = 0,
  announce = true,
  className,
}: WorldClockBoardProps) {
  const [date, setDate] = useState<Date | null>(null);

  useEffect(() => {
    const publish = () => {
      const next = new Date();
      setDate((current) => {
        if (current && clockLines(cities, current).join("\n") === clockLines(cities, next).join("\n")) {
          return current;
        }
        return next;
      });
    };

    publish();
    const timer = window.setInterval(publish, 1000);
    return () => window.clearInterval(timer);
  }, [cities]);

  const rows = clockLines(cities, date);
  const shown = useFlapRows(rows, staggerMs, cycleMs, replay);
  const label = date
    ? rows
        .map((row) => `${row.slice(0, TIME_SLOTS)} ${row.slice(TIME_SLOTS).trim()}`)
        .filter(Boolean)
        .join(". ")
    : "";

  return (
    <div
      className={`w-full overflow-hidden rounded-xl border border-white/10 bg-black px-3 py-6 shadow-[inset_0_0_16px_rgba(0,0,0,0.9)] sm:px-5 sm:py-7${className ? ` ${className}` : ""}`}
      style={{ containerType: "inline-size" }}
    >
      <p className="sr-only" aria-live={announce ? "polite" : undefined}>
        {label}
      </p>
      <div aria-hidden="true">
        <LedMatrix lines={shown} />
      </div>
    </div>
  );
}
```

Pass `cities` to change the rows. Each entry needs a label and an IANA time zone, such as `America/New_York`. `staggerMs` is the delay between shuffling blocks, and `replay` is a number you increment to shuffle the board again.

## How it works

This board is made of separate character blocks. Each block is a 5-dot-wide, 7-dot-tall picture of one character, with actual space around the block. A dot is always drawn: bright means on, dark means off. To change the display, a wave moves from left to right; each block briefly shows random characters before settling on its target.

The sample below uses the same five cities as the live board. The times are fixed here so the steps stay still while you read them.

### 01 — Start with the times as text

<WorldClockBoardStep step="0" />

The visible starting point is five lines of text: a time, a gap, and a city. `Date` represents one moment. `Intl.DateTimeFormat` is the browser API that expresses that moment in a chosen time zone. `formatToParts` gives us separate hour and minute values. `hourCycle: "h23"` gives a 24-hour clock, so every time has five characters: two digits, a colon, two digits.

```tsx title="01-clock-text.tsx"
const parts = new Intl.DateTimeFormat("en-GB", {
  timeZone,
  hour: "2-digit",
  minute: "2-digit",
  hourCycle: "h23",
}).formatToParts(date)

const hour = parts.find((part) => part.type === "hour")?.value ?? "00"
const minute = parts.find((part) => part.type === "minute")?.value ?? "00"
const time = `${hour}:${minute}`
```

**What you see:** five readable rows. New York, Tokyo, Hawaii, Los Angeles, and New Delhi each have a time in front of the name.

### 02 — Reserve five time blocks and 15 city blocks

<WorldClockBoardStep step="1" />

Every character gets the same size block, whether it is a narrow `I`, a wide `M`, or a space. The first five blocks hold `HH:MM`; the next 15 hold the city name. A CSS grid track with no LEDs separates those groups. `padEnd(15, " ")` fills unused city blocks with spaces, so `TOKYO` still occupies a 15-block field. `slice(0, 15)` keeps longer names inside it.

```tsx title="02-pad-rows.tsx"
const name = (city.name.trim().toUpperCase() || "CITY").slice(0, CITY_SLOTS)
const line = `${time}${name.padEnd(CITY_SLOTS, " ")}`
```

**What you see:** each time takes exactly five blocks. Each city begins after a real empty gap and has exactly 15 blocks, including dark ones after a short name.

### 03 — Draw every LED, including the dark ones

<WorldClockBoardStep step="2" />

The `FONT` object is a tiny bitmap font. **Bitmap** means a picture described one dot at a time. Each character maps to a string of 35 zeros and ones: seven rows × five columns. For `H`, split its string every five digits:

```text
10001  ●···●
10001  ●···●
10001  ●···●
11111  ●●●●●
10001  ●···●
10001  ●···●
10001  ●···●
```

Here `1` means a bright dot and `0` means a dark dot. We store the seven rows as one string to make lookup simple. Coordinates start at zero: `x` counts across from 0 to 4, and `y` counts down from 0 to 6. In `y * 5 + x`, multiplying by five skips the rows above; adding `x` picks the dot in the current row. For example, `(x=2, y=3)` reads position 17, the center dot of `H`'s crossbar.

Lit lamps are `#f2f2f2`. Dark lamps are `#434343`. There is no glow, so the dots stay sharp.

`H` is a useful one to read by hand: the left and right columns are on, and the middle row connects them.

```tsx title="03-glyph.tsx"
const H = "10001100011000111111100011000110001"

function lit(character: string, x: number, y: number) {
  const glyph = FONT[character] ?? FONT[" "]
  return glyph[y * 5 + x] === "1"
}
```

The board renders one little grid for each character block. `Array.from` makes 35 dots for that block. `dot % 5` gives the horizontal position, and `Math.floor(dot / 5)` gives the vertical position. The outer grid leaves space *between* blocks, and its empty track separates the time and city. Row gap separates the five clocks. No extra dark LEDs are inserted into those gaps.

```tsx title="03-separate-blocks.tsx"
<div
  className="grid items-start gap-x-[clamp(4px,0.7cqi,8px)]"
  style={{ gridTemplateColumns: `repeat(5, minmax(0, 1fr)) minmax(0, 1fr) repeat(15, minmax(0, 1fr))` }}
>
  {/* Five time blocks, an empty track, then 15 city blocks */}
</div>
```

```tsx title="03-paint-dots.tsx"
{Array.from({ length: CELL_W * CELL_H }, (_, dot) => (
  <span
    key={dot}
    className={lit(character, dot % CELL_W, Math.floor(dot / CELL_W))
      ? "bg-[#f2f2f2]"
      : "bg-[#434343]"}
  />
))}
```

**What you see:** bright dots draw `19:45` and the city name. Each 5×7 block has its own boundary of black space. Unused city blocks remain dark, but still have the same 35 dots.

### 04 — Shuffle each block, then settle it

<WorldClockBoardStep step="3" />

`performance.now()` gives a steadily increasing time in milliseconds. Each block gets a start delay based on its position. `Math.cos` shapes the delay into a smooth wave: slow at the left edge, faster through the middle, and slow again at the right edge. Before its turn, the block keeps its old character. It then shuffles random letters and digits from `CHARSET` for `HOLD_MS` milliseconds and settles on its target. The same 5×7 bitmap lookup turns each temporary character into dots.

`setInterval` calls `paint` every `cycleMs` milliseconds while blocks are shuffling. It returns a timer ID. `clearInterval` cancels that timer when all blocks settle. React also calls the effect's cleanup if the time changes again or the component leaves the page, so the previous timer cannot keep repainting it.

`matchMedia("(prefers-reduced-motion: reduce)")` reads the operating system setting for less motion. When that matches, no cell is marked dirty, the interval never starts, and the final text appears at once.

```tsx title="04-lock.tsx"
const city = column >= TIME_SLOTS
const position = city ? column - TIME_SLOTS : column
const last = city ? CITY_SLOTS - 1 : TIME_SLOTS - 1
const delay = stagger * last * (1 - Math.cos(Math.PI * position / last)) / 2
if (!dirty[rowIndex]?.[column] || now >= started + delay + HOLD_MS) {
  line += goal
} else if (now < started + delay) {
  pending = true
  line += prior?.[rowIndex]?.[column] ?? " "
} else {
  pending = true
  line += CHARSET[Math.floor(Math.random() * CHARSET.length)]
}
```

```tsx title="04-timer.tsx"
if (paint()) timer = window.setInterval(paint, cycle)
return () => window.clearInterval(timer)
```

**What you see:** blocks shuffle from left to right, then settle on the time and city.

### 05 — Update the clock and shuffle the city

<WorldClockBoardStep step="4" />

A React ref remembers the last completed rows without causing a render. Once a second, a timer makes a new `Date` and formats the times again. If the displayed text is unchanged, the state update is skipped. When a displayed minute changes, the changed time digits shuffle and **all 15 city blocks in that row** shuffle, even the blocks that will end as spaces. The gap is empty throughout.

Comparing the first five characters of the old and new row tells us whether that city's time changed. `replay` deliberately marks every time and city block for another shuffle. `matchMedia("(prefers-reduced-motion: reduce)")` reads the operating system's motion preference; when it matches, the final text appears immediately and no timer starts.

The dots are `aria-hidden`, so a screen reader does not try to read a picture of text. The paragraph next to them is the real sentence, visually hidden with `sr-only`. Leave `announce` on and that paragraph uses `aria-live="polite"`. When the label changes, assistive tech can announce the new times without interrupting what the person is doing. The tutorial previews turn `announce` off so the same times are not spoken twice.

```tsx title="05-changed-cells.tsx"
if (reduce) return false
if (force || prior === null) return true
if (column >= TIME_SLOTS && prior[rowIndex]?.slice(0, TIME_SLOTS) !== row.slice(0, TIME_SLOTS)) return true
return prior[rowIndex]?.[column] !== character
```

```tsx title="05-live-text.tsx"
<p className="sr-only" aria-live={announce ? "polite" : undefined}>
  {label}
</p>
<div aria-hidden="true">
  <LedMatrix lines={shown} />
</div>
```

**What you see:** the board holds still between changes. When a city's minute changes, its changed digits and all 15 city blocks shuffle, then the name reappears with dark blocks after it.
