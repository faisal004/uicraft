## Implementation

Copy this one component into a React app using Tailwind. The live demo above renders this exact source. It uses the site’s `--craft-surface`, `--craft-accent`, and `--font-geist-sans` variables; replace those three theme values when using it elsewhere.

```tsx title="svg-claw.tsx"
"use client";

import { useEffect, useRef, useState } from "react";

const SCENE = { dotX: 420, dotY: 200, radius: 10, grip: 62, startY: -110, restY: -110 };
const TIME = { approach: 1450, align: 1750, release: 2150, clear: 2500, retreat: 3450, secondApproach: 4200, secondAlign: 4450, secondRelease: 4850, secondClear: 5150, secondRetreat: 5900, end: 6900 };
const BAR = { y: 234, width: 62, height: 13, grip: 62 };
const ease = (value: number) => 1 - Math.pow(1 - Math.min(1, Math.max(0, value)), 3);
const between = (time: number, start: number, end: number) => ease((time - start) / (end - start));

export function SvgClawDemo() {
  const sceneRef = useRef<HTMLDivElement>(null);
  const carriageRef = useRef<SVGGElement>(null);
  const upperRef = useRef<SVGGElement>(null);
  const lowerRef = useRef<SVGGElement>(null);
  const dotRef = useRef<SVGGElement>(null);
  const guidesRef = useRef<SVGGElement>(null);
  const secondCarriageRef = useRef<SVGGElement>(null);
  const secondUpperRef = useRef<SVGGElement>(null);
  const secondLowerRef = useRef<SVGGElement>(null);
  const barRef = useRef<SVGGElement>(null);
  const teeRef = useRef<SVGTextElement>(null);
  const secondGuidesRef = useRef<SVGGElement>(null);
  const textMeasureRef = useRef<SVGTextElement>(null);
  const [teeX, setTeeX] = useState(685);
  const playheadRef = useRef(0);
  const runningRef = useRef(false);
  const frameRef = useRef(0);
  const lastRef = useRef(0);
  const [playing, setPlaying] = useState(false);

  useEffect(() => {
    document.fonts.ready.then(() => {
      if (textMeasureRef.current) setTeeX(442 + textMeasureRef.current.getComputedTextLength());
    });
  }, []);

  function paint(time: number) {
    const arrive = between(time, 0, TIME.approach);
    const leave = between(time, TIME.release, TIME.retreat);
    const targetJointY = SCENE.dotY - SCENE.grip;
    const jointY = SCENE.startY + (targetJointY - SCENE.startY) * arrive + (SCENE.restY - targetJointY) * leave;
    const opening = between(time, TIME.align, TIME.release) * 30;
    const closing = between(time, TIME.clear, TIME.retreat) * 30;
    const angle = opening - closing;
    const dotY = time < TIME.align ? jointY + SCENE.grip : SCENE.dotY;
    const guideOpacity = Math.min(between(time, 750, TIME.approach), 1 - between(time, TIME.release, TIME.retreat)) * 0.3;
    carriageRef.current?.setAttribute("transform", `translate(${SCENE.dotX} ${jointY}) rotate(90)`);
    upperRef.current?.setAttribute("transform", `rotate(${-angle})`);
    lowerRef.current?.setAttribute("transform", `rotate(${angle})`);
    dotRef.current?.setAttribute("transform", `translate(${SCENE.dotX} ${dotY})`);
    guidesRef.current?.setAttribute("opacity", String(guideOpacity));

    const barRight = teeX + BAR.width;
    const secondTarget = barRight + BAR.grip;
    const secondArrive = between(time, TIME.retreat, TIME.secondApproach);
    const secondLeave = between(time, TIME.secondRelease, TIME.secondRetreat);
    const secondX = 1100 + (secondTarget - 1100) * secondArrive + (1100 - secondTarget) * secondLeave;
    const secondOpening = between(time, TIME.secondAlign, TIME.secondRelease) * 30;
    const secondClosing = between(time, TIME.secondClear, TIME.secondRetreat) * 30;
    const secondAngle = secondOpening - secondClosing;
    const barX = time < TIME.secondAlign ? secondX - BAR.grip - BAR.width : teeX;
    const secondGuideOpacity = Math.min(between(time, TIME.secondApproach - 500, TIME.secondApproach), 1 - between(time, TIME.secondRelease, TIME.secondRetreat)) * 0.3;
    secondCarriageRef.current?.setAttribute("transform", `translate(${secondX} ${BAR.y + BAR.height / 2}) scale(-1 1)`);
    secondUpperRef.current?.setAttribute("transform", `rotate(${-secondAngle})`);
    secondLowerRef.current?.setAttribute("transform", `rotate(${secondAngle})`);
    barRef.current?.setAttribute("transform", `translate(${barX - teeX} 0)`);
    barRef.current?.setAttribute("opacity", time < TIME.secondAlign ? "1" : "0");
    if (time < TIME.secondAlign) teeRef.current?.setAttribute("mask", "url(#t-stem-mask)");
    else teeRef.current?.removeAttribute("mask");
    secondGuidesRef.current?.setAttribute("opacity", String(secondGuideOpacity));
  }

  function stop() {
    runningRef.current = false;
    cancelAnimationFrame(frameRef.current);
    setPlaying(false);
  }

  function tick(now: number) {
    if (!runningRef.current) return;
    playheadRef.current = Math.min(TIME.end, playheadRef.current + now - lastRef.current);
    lastRef.current = now;
    paint(playheadRef.current);
    if (playheadRef.current >= TIME.end) stop();
    else frameRef.current = requestAnimationFrame(tick);
  }

  function play() {
    if (playheadRef.current >= TIME.end) playheadRef.current = 0;
    if (runningRef.current) return;
    runningRef.current = true;
    lastRef.current = performance.now();
    setPlaying(true);
    frameRef.current = requestAnimationFrame(tick);
  }

  function replay() {
    stop();
    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    playheadRef.current = reduced ? TIME.end : 0;
    paint(playheadRef.current);
    if (!reduced) play();
  }

  useEffect(() => {
    const motion = window.matchMedia("(prefers-reduced-motion: reduce)");
    if (motion.matches) {
      playheadRef.current = TIME.end;
      paint(TIME.end);
      return;
    }
    paint(0);
    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        observer.disconnect();
        play();
      }
    }, { threshold: 0.35 });
    if (sceneRef.current) observer.observe(sceneRef.current);
    return () => {
      observer.disconnect();
      runningRef.current = false;
      cancelAnimationFrame(frameRef.current);
    };
    // The observer owns this measured scene; playback uses its current closures.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [teeX]);

  const buttonClass = "rounded-lg px-3 py-1.5 text-sm text-foreground/65 hover:bg-foreground/5 hover:text-foreground focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-(--craft-accent)";

  return (
    <div ref={sceneRef} className="not-prose overflow-hidden border border-foreground/10 bg-(--craft-surface) text-foreground">
      <svg className="block w-full" viewBox="0 0 1000 500" role="img" aria-labelledby="claw-title claw-description">
        <title id="claw-title">Build it — SVG claw</title>
        <desc id="claw-description">Two mechanical claws complete “build it”: one places the dot of the first i, and the other places the crossbar of the final t.</desc>
        <g fill="currentColor" fontFamily="var(--font-geist-sans), Arial, sans-serif" fontSize="146" fontWeight="800" letterSpacing="-5">
          <text x="230" y="310">bu</text>
          <rect x="410" y="238" width="20" height="72" rx="1" />
          <text x="442" y="310">ld i</text>
          <text ref={teeRef} x={teeX} y="310" mask="url(#t-stem-mask)">t</text>
        </g>
        <defs>
          <mask id="t-stem-mask" maskUnits="userSpaceOnUse" x="0" y="0" width="1000" height="500">
            <rect width="1000" height="500" fill="white" />
            <rect x={teeX - 10} y="231" width="100" height="19" fill="black" />
          </mask>
          <clipPath id="t-crossbar"><rect x={teeX - 10} y="231" width="100" height="19" /></clipPath>
        </defs>
        <text ref={textMeasureRef} x="442" y="310" visibility="hidden" aria-hidden="true" fill="currentColor" fontFamily="var(--font-geist-sans), Arial, sans-serif" fontSize="146" fontWeight="800" letterSpacing="-5">ld i</text>
        <g ref={guidesRef} opacity="0" aria-hidden="true" fill="none" stroke="currentColor" strokeWidth="1.5">
          <path d={`M ${SCENE.dotX} 100 V 285`} strokeDasharray="5 7" />
          <path d={`M ${SCENE.dotX - 20} ${SCENE.dotY} H ${SCENE.dotX + 20} M ${SCENE.dotX} ${SCENE.dotY - 20} V ${SCENE.dotY + 20}`} />
        </g>
        <g ref={secondGuidesRef} opacity="0" aria-hidden="true" fill="none" stroke="currentColor" strokeWidth="1.5">
          <path d={`M ${teeX + BAR.width / 2} 165 V 305`} strokeDasharray="5 7" />
          <path d={`M ${teeX - 15} ${BAR.y + BAR.height / 2} H ${teeX + BAR.width + 15}`} />
        </g>
        <g ref={carriageRef} fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round">
          <g aria-hidden="true">
            <path d="M -500 0 H 0" />
            <g ref={upperRef}><path d="M 0 0 L 28 -25 L 56 -7" /></g>
            <g ref={lowerRef}><path d="M 0 0 L 28 25 L 56 7" /></g>
            <circle r="9" fill="var(--craft-surface)" />
          </g>
        </g>
        <g ref={dotRef} fill="currentColor"><circle r={SCENE.radius} /></g>
        <g ref={barRef} fill="currentColor" fontFamily="var(--font-geist-sans), Arial, sans-serif" fontSize="146" fontWeight="800" letterSpacing="-5"><text x={teeX} y="310" clipPath="url(#t-crossbar)">t</text></g>
        <g ref={secondCarriageRef} fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
          <path d="M -500 0 H 0" />
          <g ref={secondUpperRef}><path d="M 0 0 L 28 -25 L 56 -7" /></g>
          <g ref={secondLowerRef}><path d="M 0 0 L 28 25 L 56 7" /></g>
          <circle r="9" fill="var(--craft-surface)" />
        </g>
      </svg>
      <div className="flex items-center justify-center gap-2 border-t border-foreground/10 px-4 py-3">
        <button type="button" className={buttonClass} onClick={replay}>Replay</button>
        <button type="button" className={buttonClass} onClick={() => runningRef.current ? stop() : play()}>
          {playing ? "Pause" : "Play"}
        </button>
      </div>
    </div>
  );
}
```

## How it works

### 01 — Draw the incomplete letters

The SVG `viewBox` defines one 1000 × 500 coordinate system. The phrase uses the site’s bold sans-serif font. A small rectangle makes the first `i` stem. The final `t` is the real font glyph split into two parts: a mask removes only its crossbar band from the stationary copy, while a matching clip shows that same band on the moving copy. At alignment, the component reveals the complete original glyph and hides the moving copy. This makes the finished `t` continuous even at fractional SVG pixel positions.

```tsx
<text x="442" y="310">ld i</text>
<text x={teeX} y="310" mask="url(#t-stem-mask)">t</text>
<g ref={barRef}><text x={teeX} y="310" clipPath="url(#t-crossbar)">t</text></g>
```

### 02 — Measure the `t`

After `document.fonts.ready`, `getComputedTextLength()` measures the preceding `ld i` text in SVG units. That places the actual `t` immediately after it. The second claw’s destination and the crossbar guide use the same `teeX` coordinate.

```tsx
const teeX = 442 + textMeasureRef.current.getComputedTextLength();
```

### 03 — Hang the first claw from above

The first carriage translates vertically, then rotates its local horizontal shaft by 90 degrees. Its shaft therefore extends above the visible scene. Two nested finger groups rotate around the same joint. The dot starts between their tips and follows the joint downward at a fixed grip offset.

```tsx
carriageRef.current?.setAttribute("transform", `translate(${SCENE.dotX} ${jointY}) rotate(90)`);
const dotY = time < TIME.align ? jointY + SCENE.grip : SCENE.dotY;
```

When the fingers open, the dot stops following the claw. The claw lifts fully out of the SVG while the dot remains above the `i`.

### 04 — Bring in the second claw

A mirrored carriage enters from the right carrying the clipped crossbar. Its finger groups use the same joint geometry as the first claw. While held, the crossbar follows the carriage; at alignment, the complete `t` replaces the split copies while the fingers open and leave.

```tsx
secondCarriageRef.current?.setAttribute("transform", `translate(${secondX} ${BAR.y + BAR.height / 2}) scale(-1 1)`);
barRef.current?.setAttribute("transform", `translate(${barX - teeX} 0)`);
```

Try changing `TIME.secondApproach`. Watch the gap between the two placements change without changing either final letter.

### 05 — Run and clean up one timeline

`requestAnimationFrame` advances one playhead using elapsed milliseconds. `paint` derives both carriage positions, finger angles, and independent letter pieces from that time, so the claws cannot drift out of sync. `IntersectionObserver` starts the scene when visible. Replay resets the playhead; Pause cancels the next frame; Play resumes. Cleanup disconnects the observer and cancels the frame. Reduced motion displays both completed letters immediately.
